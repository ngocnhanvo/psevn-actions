name: NCKH dispatch one targeted staging workflow (no GitHub issue fan-out)
on:
  push:
    branches: [main]
    paths:
      - 'docs/nckh-workflow-request.json'
permissions:
  contents: read
  actions: write
jobs:
  dispatch:
    if: github.repository == 'ngocnhanvo/nghiencuukhoahoc-actions' && github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    timeout-minutes: 5
    steps:
      - uses: actions/checkout@v4
        with:
          persist-credentials: false
      - name: Dispatch exactly one approved NCKH workflow
        env:
          GH_TOKEN: ${{ github.token }}
        shell: bash
        run: |
          set -euo pipefail
          set +x
          python3 - <<'PY'
          import json,os,pathlib,re,urllib.request,urllib.error
          repo='ngocnhanvo/nghiencuukhoahoc-actions'
          if os.environ.get('GITHUB_REPOSITORY')!=repo or os.environ.get('GITHUB_REF')!='refs/heads/main':
              raise RuntimeError('Wrong NCKH repository or branch')
          doc=json.loads(pathlib.Path('docs/nckh-workflow-request.json').read_text(encoding='utf8'))
          allow={
              'php-ci':'nckh-php-offline-check.yml',
              'astro-ci':'nckh-notebook-feature-check.yml',
              'astro-public-cache-ci':'nckh-astro-ci-check.yml',
              'astro-build-readonly':'nckh-astro-source-build-check.yml',
              'php-stage':'nckh-ai-staging-deploy.yml',
              'research-stage':'nckh-notebook-staging-deploy.yml',
              'astro-stage':'nckh-astro-github-staging-deploy.yml',
          }
          task=doc.get('task')
          request_id=doc.get('request_id','')
          if task not in allow or not isinstance(request_id,str) or not re.fullmatch('[A-Za-z0-9_-]{8,80}',request_id):
              raise RuntimeError('Rejected unsafe or invalid workflow request')
          wf=allow[task]
          token=os.environ.get('GH_TOKEN','')
          if not token:raise RuntimeError('Missing scoped GitHub Actions token')
          req=urllib.request.Request(
              'https://api.github.com/repos/'+repo+'/actions/workflows/'+wf+'/dispatches',
              data=b'{"ref":"main"}',method='POST',headers={
                  'Authorization':'Bearer '+token,'Accept':'application/vnd.github+json',
                  'X-GitHub-Api-Version':'2022-11-28',
                  'Content-Type':'application/json','User-Agent':'NCKH-OneTask-Dispatcher',
              })
          try:
              with urllib.request.urlopen(req,timeout=25) as response:
                  if response.status!=204:raise RuntimeError('Dispatch did not receive HTTP 204')
          except urllib.error.HTTPError as error:
              raise RuntimeError('Scoped workflow dispatch rejected, HTTP '+str(error.code)) from None
          report='PASS: sent one scoped '+task+' dispatch on Actions main; request '+request_id
          print(report)
          with open(os.environ['GITHUB_STEP_SUMMARY'],'a',encoding='utf8') as out:
              out.write('### '+report+'\nNo issue event, no FTP or production changes in dispatcher.\n')
          PY
