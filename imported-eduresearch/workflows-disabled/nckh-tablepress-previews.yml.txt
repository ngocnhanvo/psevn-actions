name: Update NCKH ESC Table Preview Images
on:
  workflow_dispatch:
    inputs:
      mode:
        description: 'missing: only missing previews, single: one ID, selected: several IDs'
        required: true
        default: missing
        type: choice
        options: [missing, single, selected]
      shortcodes:
        description: 'Comma-separated ESC Table IDs; blank scans all in missing mode'
        required: false
        default: ''
        type: string
      force:
        description: 'Replace existing screenshots even when already mapped'
        required: false
        default: false
        type: boolean
      page_url:
        description: 'Optional full URL of one NCKH production page'
        required: false
        default: ''
        type: string
permissions:
  contents: read
concurrency:
  group: nckh-tablepress-previews
  cancel-in-progress: false
jobs:
  previews:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    env:
      WP_ADMIN_URL: https://api-nghiencuukhoahoc.webnvn.com
      TABLEPRESS_PUBLIC_ORIGIN: https://nghiencuukhoahoc.id.vn
      WP_ADMIN_USERNAME: ${{ secrets.WP_ADMIN_USERNAME }}
      WP_ADMIN_PASSWORD: ${{ secrets.WP_ADMIN_PASSWORD }}
      SOURCE_REPO_PAT: ${{ secrets.SOURCE_REPO_PAT }}
      TABLEPRESS_PREVIEW_MODE: ${{ inputs.mode }}
      TABLEPRESS_SHORTCODES: ${{ inputs.shortcodes }}
      TABLEPRESS_FORCE: ${{ inputs.force }}
      TABLEPRESS_PAGE_URL: ${{ inputs.page_url }}
    steps:
      - name: Validate NCKH source and account secrets
        shell: bash
        run: |
          set -euo pipefail
          test -n "$SOURCE_REPO_PAT"
          test -n "$WP_ADMIN_USERNAME"
          test -n "$WP_ADMIN_PASSWORD"
          test "$WP_ADMIN_URL" = "https://api-nghiencuukhoahoc.webnvn.com"
          test "$TABLEPRESS_PUBLIC_ORIGIN" = "https://nghiencuukhoahoc.id.vn"
      - name: Checkout NCKH PHP source only
        uses: actions/checkout@v4
        with:
          repository: ngocnhanvo/nghiencuukhoahoc-php
          ref: master
          token: ${{ secrets.SOURCE_REPO_PAT }}
          path: source
          persist-credentials: false
      - name: Verify screenshot automation source
        shell: bash
        run: |
          set -euo pipefail
          node --check source/automation/tablepress/previews.mjs
          grep -Fq 'nckh-tablepress/v1/preview' source/automation/tablepress/previews.mjs
          grep -Fq 'api-nghiencuukhoahoc.webnvn.com' source/automation/tablepress/previews.mjs
          if grep -qi 'xuongmunonbaohiem' source/automation/tablepress/previews.mjs; then
            echo 'ERROR: legacy Blue Sea domain present'; exit 1
          fi
      - name: Set up Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '24'
      - name: Install Chromium and screenshot dependencies
        working-directory: source
        run: |
          npm install --no-save playwright@1.55.0 sharp@0.34.3
          npx playwright install --with-deps chromium
      - name: Capture only published NCKH frontend sections and save page-scoped previews
        working-directory: source
        run: node automation/tablepress/previews.mjs
      - name: Upload preview diagnostics
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: nckh-tablepress-previews
          path: source/tablepress-previews/
          if-no-files-found: ignore
          retention-days: 7
