name: NCKH audit noisy Actions and latest deploy (read-only)
on:
  push:
    branches: [main]
    paths:
      - '.github/workflows/nckh-actions-run-audit.yml'
  workflow_dispatch:
permissions:
  contents: read
  actions: read
  issues: write
jobs:
  audit:
    if: github.repository == 'ngocnhanvo/nghiencuukhoahoc-actions' && github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - name: Checkout only this project's Actions configuration
        uses: actions/checkout@v4
        with:
          persist-credentials: false
      - name: Inventory issue-triggered workflows and current completed/active run statuses
        env:
          GH_TOKEN: ${{ github.token }}
        shell: bash
        run: |
          set -euo pipefail
          set +x
          python3 - <<'PY'
          import json,os,pathlib,time,urllib.request,urllib.error
          root='https://api.github.com/repos/ngocnhanvo/nghiencuukhoahoc-actions'
          token=os.environ.get('GH_TOKEN','')
          if os.environ.get('GITHUB_REPOSITORY')!='ngocnhanvo/nghiencuukhoahoc-actions' or not token:
              raise RuntimeError('This read-only audit is restricted to the NCKH Actions repo')
          def api(method,path,data=None):
              if not path.startswith('/repos/ngocnhanvo/nghiencuukhoahoc-actions/'):
                  raise RuntimeError('Disallowed repository path')
              request=urllib.request.Request('https://api.github.com'+path,method=method,
                  data=(json.dumps(data).encode('utf8') if data is not None else None),
                  headers={'Authorization':'Bearer '+token,'Accept':'application/vnd.github+json',
                           'X-GitHub-Api-Version':'2022-11-28',
                           'User-Agent':'NCKH-Actions-ReadOnly-Audit',
                           **({'Content-Type':'application/json'} if data is not None else {})})
              with urllib.request.urlopen(request,timeout=25) as r:
                  body=r.read(2500000)
                  return json.loads(body) if body else None
          flows=list(pathlib.Path('.github/workflows').glob('*.y*ml'))
          issues_listening=[p.name for p in flows if
              ('  issues:' in ('\n'+p.read_text(encoding='utf8')))]
          titles={
              '[nckh] Verify backend offline only':'.github/workflows/nckh-php-offline-check.yml',
              '[nckh] Verify notebook source only':'.github/workflows/nckh-notebook-feature-check.yml',
              '[nckh] Deploy consented AI API to staging':'.github/workflows/nckh-ai-staging-deploy.yml',
              '[nckh] Deploy Astro github to staging':'.github/workflows/nckh-astro-github-staging-deploy.yml'
          }
          found={key:[] for key in titles}
          cleanup_runs=[]
          total=None
          skipped=0
          seen=0
          active=0
          for page in range(1,26):
              payload=api('GET','/repos/ngocnhanvo/nghiencuukhoahoc-actions/actions/runs?per_page=100&page='+str(page))
              if total is None:total=payload.get('total_count')
              rows=payload.get('workflow_runs') or []
              if not rows:break
              for run in rows:
                  seen+=1
                  status=run.get('status')
                  if status!='completed':active+=1
                  if status=='completed' and run.get('conclusion')=='skipped':skipped+=1
                  title=run.get('display_title')
                  source=run.get('path','').replace('@main','')
                  if source.endswith('nckh-delete-skipped-runs.yml'):
                      cleanup_runs.append(run)
                  if (title in titles and source.endswith(titles[title])
                      and run.get('created_at','')>='2026-10-20T00:30:00Z'):
                      found[title].append(run)
              if len(rows)<100:break
          cleanup_runs.sort(key=lambda r:r.get('created_at',''),reverse=True)
          latest_cleaner=cleanup_runs[0] if cleanup_runs else None
          cleaner_line=('Cleanup workflow: [run '+str(latest_cleaner.get('id'))+']('+str(latest_cleaner.get('html_url'))+') — '+str(latest_cleaner.get('status'))+'/'+str(latest_cleaner.get('conclusion')) if latest_cleaner else 'Cleanup workflow run: not found in recent 2,500 runs')
          report=[
              '## NCKH Actions audit (automatic read-only report)',
              cleaner_line,
              'Total runs currently reported by API: **'+str(total)+'**. Scanned: **'+str(seen)+'**; skipped completed: **'+str(skipped)+'**; noncompleted: **'+str(active)+'**.',
              'Tracked workflow configuration files: **'+str(len(flows))+'**; workflows listening for `issues`: **'+str(len(issues_listening))+'**.',
              '',
              '| Requested task | Matching workflow run (latest) | Conclusion |',
              '|---|---|---|'
          ]
          for title,rows in found.items():
              rows.sort(key=lambda r:r.get('created_at',''),reverse=True)
              if rows:
                  r=rows[0]
                  conclusion=r.get('conclusion') or r.get('status') or 'unknown'
                  url=r.get('html_url','')
                  label='[Run '+str(r.get('id'))+']('+url+')'
              else:
                  label='Not found in scanned recent runs'
                  conclusion='unknown'
              report.append('| '+title.replace('|','/')+' | '+label+' | '+str(conclusion)+' |')
          report.extend([
              '',
              'This audit **does not delete runs or deploy code**. The screenshot clutter is caused by workflows reacting to every new issue and then being skipped.',
              'Workflow paths with issue listeners (sample): '+', '.join(issues_listening[:15]),
              'The 2026-10-19 legacy cleanup cutoff does not cover new issue-triggered runs.'
          ])
          body='\n'.join(report)
          print(body)
          with open(os.environ['GITHUB_STEP_SUMMARY'],'a',encoding='utf8') as f:f.write(body+'\n')
          try:
              api('POST','/repos/ngocnhanvo/nghiencuukhoahoc-actions/issues/302/comments',
                  {'body':body[:62000]})
              print('Audit report posted to existing issue #302 for independent inspection')
          except urllib.error.HTTPError as exc:
              print('Could not post audit comment; GitHub HTTP',exc.code)
              raise
          PY
