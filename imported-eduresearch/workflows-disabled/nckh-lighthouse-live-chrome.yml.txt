name: NCKH Lighthouse real staging Chromium smoke
on:
  issues:
    types: [opened, reopened]
permissions:
  contents: read
jobs:
  browser:
    if: >-
      github.repository == 'ngocnhanvo/nghiencuukhoahoc-actions'
      && github.event.issue.user.login == 'ngocnhanvo'
      && github.event.issue.title == '[nckh] Lighthouse staging Chrome qa'
    runs-on: ubuntu-latest
    timeout-minutes: 12
    steps:
      - uses: actions/checkout@v4
        with:
          persist-credentials: false
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - name: Install isolated Chromium
        shell: bash
        run: |
          set -euo pipefail
          python -m pip install --disable-pip-version-check 'playwright>=1.50,<2'
          python -m playwright install --with-deps chromium
      - name: Read-only staging homepage WCAG and static SEO endpoint checks
        shell: bash
        run: |
          set -euo pipefail
          python scripts/nckh_lighthouse_live_chrome.py
          echo 'PASS: real staging Chromium homepage, llms.txt, robots.txt, sitemap.xml' >> "$GITHUB_STEP_SUMMARY"
          echo 'READ-ONLY: no staging writes, no production deployment.' >> "$GITHUB_STEP_SUMMARY"
