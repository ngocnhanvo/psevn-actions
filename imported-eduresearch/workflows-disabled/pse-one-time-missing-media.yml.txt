    if: github.repository == 'ngocnhanvo/psevn-actions'

          FTP_USERNAME: ${{ secrets.FTP_USERNAME }}
          FTP_PASSWORD: ${{ secrets.FTP_PASSWORD }}

          set -euo pipefail
          python3 -m py_compile scripts/pse-one-time-missing-media.py
