      - name: Stage exactly one guarded WordPress MU plugin with FTP-TLS backup
        shell: bash
        run: |
          set -euo pipefail
          set +x
          python3 - <<'PY'
          import os,ftplib,hashlib,io,time,sys
          from urllib.parse import urlparse
          src='source/wp-content/mu-plugins/nckh-research-api.php'
          target='public_html/wp-content/mu-plugins/nckh-research-api.php'
          base='/domains/api-nghiencuukhoahoc.webnvn.com'
          content=open(src,'rb').read()
          assert b"private const HOST = 'api-nghiencuukhoahoc.webnvn.com'" in content
          assert b'Plugin Name: EduResearch Hub' in content
          assert b'get_current_user_id()' in content
          for k in ('FTP_SERVER','FTP_USERNAME','FTP_PASSWORD'):
              if not os.environ.get(k): sys.exit('STOP: Missing staging deployment secret '+k)
          endpoint=os.environ['FTP_SERVER']
          u=urlparse(endpoint if '://' in endpoint else 'ftps://'+endpoint)
          ftp=ftplib.FTP_TLS()
          ftp.connect(u.hostname,u.port or 21,timeout=22)
          ftp.login(os.environ['FTP_USERNAME'],os.environ['FTP_PASSWORD'])
          ftp.prot_p()
          ftp.cwd(base)
          if 'api-nghiencuukhoahoc.webnvn.com' not in ftp.pwd():
              sys.exit('STOP: Incorrect FTP working directory')
          stamp=str(int(time.time()))
          backup='nckh-config-backups'
          try: ftp.mkd(backup)
          except ftplib.error_perm as e:
              if not str(e).startswith('550'): raise
          staged=f'{backup}/nckh-research-api-stage-{stamp}.php.txt'
          oldpath=f'{backup}/nckh-research-api-old-{stamp}.php.txt'
          old=None
          try:
              buff=io.BytesIO()
              ftp.retrbinary('RETR '+target,buff.write)
              old=buff.getvalue()
          except ftplib.error_perm as e:
              if not str(e).startswith('550'):raise
          digest=hashlib.sha256(content).digest()
          if old is not None and hashlib.sha256(old).digest()==digest:
              print('PASS: NCKH staging API already matches the requested source; no upload needed.')
              ftp.quit()
              sys.exit(0)
          ftp.storbinary('STOR '+staged,io.BytesIO(content))
          verify=io.BytesIO();ftp.retrbinary('RETR '+staged,verify.write)
          if hashlib.sha256(verify.getvalue()).digest()!=digest:
              raise RuntimeError('Staging upload hash mismatch')
          renamed=False
          try:
              if old is not None:
                  ftp.rename(target,oldpath)
                  renamed=True
              ftp.rename(staged,target)
              current=io.BytesIO();ftp.retrbinary('RETR '+target,current.write)
              if hashlib.sha256(current.getvalue()).digest()!=digest:
                  raise RuntimeError('Published staging file checksum mismatch')
              print('PASS: NCKH WordPress MU plugin deployed to approved staging API host only.')
              if renamed:print('Original MU plugin retained outside web root for rollback.')
          except Exception:
              try:ftp.delete(target)
              except ftplib.all_errors:pass
              if renamed: ftp.rename(oldpath,target)
              raise
          finally:
              try:ftp.quit()
              except ftplib.all_errors:pass
          PY

          try:
              with urllib.request.urlopen(req,timeout=20) as response:
                  status=response.status;body=response.read(512)
          except urllib.error.HTTPError as e:
              status=e.code;body=e.read(512)
          try: data=json.loads(body)
          except ValueError: data={}

          if status not in (401,403) or not isinstance(data,dict) or not data.get('code'):
              raise SystemExit('FAIL: Private REST endpoint not available/secure on staging')
